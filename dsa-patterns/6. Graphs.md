### 2. Questions to Ask Before You Code

1. Is the graph directed or undirected?    
2. Is it weighted or unweighted?
3. Are there negative weights? 
4. What representation should I use? (adjacency list/matrix)
5. Is the problem asking about:
    - Connectivity?
    - Shortest/Longest Path?
    - Cycles?
    - Components?
    - Topological order?

## 4. Code Templates (Simplified)

1. BFS Template (Unweighted Shortest Path)

```cpp
queue<pair<int, int>> q; // node, distance
unordered_set<int> visited;

q.push({src, 0});
visited.insert(src);

while (!q.empty()) {
    auto [node, dist] = q.front(); q.pop();
    for (int nei : graph[node]) {
        if (!visited.count(nei)) {
            q.push({nei, dist + 1});
            visited.insert(nei);
        }
    }
}
```

2. DFS Template

```cpp
void dfs(int node, unordered_set<int>& visited) {
    visited.insert(node);
    for (int nei : graph[node]) {
        if (!visited.count(nei))
            dfs(nei, visited);
    }
}
```

3. Dijkstra’s Algorithm

```cpp
priority_queue<pair<int, int>, vector<pair<int, int>>, greater<>> pq;
vector<int> dist(n, INT_MAX);

pq.push({0, start});
dist[start] = 0;

while (!pq.empty()) {
    auto [d, node] = pq.top(); pq.pop();
    for (auto [nei, wt] : graph[node]) {
        if (dist[nei] > d + wt) {
            dist[nei] = d + wt;
            pq.push({dist[nei], nei});
        }
    }
}
```

4. Topological Sort (Kahn’s Algorithm)

```cpp
vector<int> indegree(n, 0);
for (auto &[u, v] : edges) indegree[v]++;
queue<int> q;
for (int i = 0; i < n; ++i) if (indegree[i] == 0) q.push(i);

vector<int> topo;
while (!q.empty()) {
    int node = q.front(); q.pop();
    topo.push_back(node);
    for (int nei : graph[node]) {
        if (--indegree[nei] == 0) q.push(nei);
    }
}
```

5. Union-Find (DSU)

```cpp
vector<int> parent(n);
iota(parent.begin(), parent.end(), 0);

int find(int x) {
    return parent[x] == x ? x : parent[x] = find(parent[x]);
}

bool unite(int u, int v) {
    int pu = find(u), pv = find(v);
    if (pu == pv) return false;
    parent[pu] = pv;
    return true;
}
```

6. DFS-based Topological Sort (stack)

```cpp
void dfs(int node, vector<bool>& visited, stack<int>& st) {
    visited[node] = true;
    for (int nei : graph[node])
        if (!visited[nei]) dfs(nei, visited, st);
    st.push(node); // push after all descendants processed
}
// pop stack top-to-bottom for topo order
```

7. Bellman-Ford (handles negative weights, detects negative cycles)

```cpp
vector<int> dist(n, INT_MAX);
dist[src] = 0;

// relax all edges n-1 times
for (int i = 0; i < n - 1; ++i) {
    for (auto &[u, v, wt] : edges) {
        if (dist[u] != INT_MAX && dist[u] + wt < dist[v])
            dist[v] = dist[u] + wt;
    }
}

// nth pass: if any edge still relaxes, negative cycle exists
bool hasNegativeCycle = false;
for (auto &[u, v, wt] : edges) {
    if (dist[u] != INT_MAX && dist[u] + wt < dist[v])
        hasNegativeCycle = true;
}
```

8. Floyd-Warshall (all-pairs shortest path, DP)

```cpp
// dist[i][j] initialized: 0 if i==j, wt if edge exists, else INF
for (int k = 0; k < n; ++k)
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < n; ++j)
            if (dist[i][k] != INF && dist[k][j] != INF)
                dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]);

// negative cycle check: dist[i][i] < 0 for any i
```

9. Prim's Algorithm (MST — grow from a node)

```cpp
priority_queue<pair<int,int>, vector<pair<int,int>>, greater<>> pq; // {wt, node}
vector<bool> inMST(n, false);
int mstWeight = 0;

pq.push({0, 0}); // start from node 0
while (!pq.empty()) {
    auto [wt, node] = pq.top(); pq.pop();
    if (inMST[node]) continue;
    inMST[node] = true;
    mstWeight += wt;
    for (auto [nei, w] : graph[node])
        if (!inMST[nei]) pq.push({w, nei});
}
```

10. Kruskal's Algorithm (MST — sort edges + DSU)

```cpp
// edges as {wt, u, v}, uses find/unite from Union-Find above
sort(edges.begin(), edges.end()); // by weight

int mstWeight = 0, edgesUsed = 0;
for (auto &[wt, u, v] : edges) {
    if (unite(u, v)) { // no cycle -> add edge
        mstWeight += wt;
        edgesUsed++;
        if (edgesUsed == n - 1) break;
    }
}
```

11. Bridges (Tarjan's — critical connections)

```cpp
vector<int> disc(n, -1), low(n, -1);
vector<vector<int>> bridges;
int timer = 0;

void dfs(int node, int parent, vector<bool>& visited) {
    visited[node] = true;
    disc[node] = low[node] = timer++;
    for (int nei : graph[node]) {
        if (nei == parent) continue;
        if (!visited[nei]) {
            dfs(nei, node, visited);
            low[node] = min(low[node], low[nei]);
            if (low[nei] > disc[node]) // no back edge to node or above
                bridges.push_back({node, nei});
        } else {
            low[node] = min(low[node], disc[nei]); // back edge
        }
    }
}
```

12. Articulation Points (Tarjan's — cut vertices)

```cpp
vector<int> disc(n, -1), low(n, -1);
vector<bool> isAP(n, false);
int timer = 0;

void dfs(int node, int parent, vector<bool>& visited) {
    visited[node] = true;
    disc[node] = low[node] = timer++;
    int children = 0;
    for (int nei : graph[node]) {
        if (nei == parent) continue;
        if (!visited[nei]) {
            children++;
            dfs(nei, node, visited);
            low[node] = min(low[node], low[nei]);
            // case 1: node is not root, and subtree can't reach above node
            if (parent != -1 && low[nei] >= disc[node])
                isAP[node] = true;
        } else {
            low[node] = min(low[node], disc[nei]);
        }
    }
    // case 2: root with 2+ DFS children
    if (parent == -1 && children > 1)
        isAP[node] = true;
}
```

13. Kosaraju's Algorithm (Strongly Connected Components)

```cpp
// Step 1: DFS on original graph, push nodes to stack by finish time
void dfs1(int node, vector<bool>& visited, stack<int>& st) {
    visited[node] = true;
    for (int nei : graph[node])
        if (!visited[nei]) dfs1(nei, visited, st);
    st.push(node);
}

// Step 2: DFS on transposed graph in stack order -> each DFS tree = one SCC
void dfs2(int node, vector<bool>& visited, vector<int>& component) {
    visited[node] = true;
    component.push_back(node);
    for (int nei : transposeGraph[node])
        if (!visited[nei]) dfs2(nei, visited, component);
}

// main:
stack<int> st;
vector<bool> visited(n, false);
for (int i = 0; i < n; ++i)
    if (!visited[i]) dfs1(i, visited, st);

fill(visited.begin(), visited.end(), false);
vector<vector<int>> sccs;
while (!st.empty()) {
    int node = st.top(); st.pop();
    if (!visited[node]) {
        vector<int> component;
        dfs2(node, visited, component);
        sccs.push_back(component);
    }
}
```

## 5. Tips to Succeed in Graph Problems

- Always draw small graph examples
- Choose the right representation: adjacency list is best for space
- For weighted graphs → Dijkstra (no negative edges), Bellman-Ford (negative edges allowed)
- Always use a visited set/array to prevent infinite loops
- Topological sort only works on DAGs
- For DSU problems: if "cycle" or "connected components" is mentioned, try Union-Find
